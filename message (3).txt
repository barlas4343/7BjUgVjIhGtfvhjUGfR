function utility:RenderObject(RenderType, RenderProperties, RenderHidden)
    local Render = Instance.new(RenderType)
    if RenderProperties and typeof(RenderProperties) == "table" then
        for Property, Value in pairs(RenderProperties) do
            if Property ~= "RenderTime" then
                Render[Property] = Value
            end
        end
    end
    library.Renders[#library.Renders + 1] = {Render, RenderProperties, RenderHidden, RenderProperties["RenderTime"] or nil}
    return Render
end
function utility:CreateConnection(ConnectionType, ConnectionCallback)
    local Connection = ConnectionType:Connect(ConnectionCallback)
    library.Connections[#library.Connections + 1] = Connection
    return Connection
end

function utility:Serialise(Table)
    local Serialised = ""
    for Index, Value in pairs(Table) do
        Serialised = Serialised .. Value .. ", "
    end
    return Serialised:sub(0, #Serialised - 2)
end
function utility:Sort(Table1, Table2)
    local Table3 = {}
    for Index, Value in pairs(Table2) do
        if table.find(Table1, Index) then
            Table3[#Table3 + 1] = Value
        end
    end
    return Table3
end

local function createResizeHandle(pos, size, directions)
    local handle = utility:RenderObject("TextButton", {
        BackgroundTransparency = 1,
        Text = "",
        Position = pos,
        Size = size,
        ZIndex = 10,
        Parent = ScreenGui_MainFrame
    })
    local resizing = false
    local resizeStart = nil
    local startWidth = nil
    local startHeight = nil
    local startX = nil
    local startY = nil
    handle.MouseButton1Down:Connect(function()
        resizing = true
        resizeStart = utility:MouseLocation()
        startWidth = ScreenGui_MainFrame.Size.X.Offset
        startHeight = ScreenGui_MainFrame.Size.Y.Offset
        startX = ScreenGui_MainFrame.Position.X.Offset
        startY = ScreenGui_MainFrame.Position.Y.Offset
    end)
    utility:CreateConnection(uis.InputChanged, function(input)
        if resizing and input.UserInputType == Enum.UserInputType.MouseMovement then
            local delta = utility:MouseLocation() - resizeStart
            local newW = startWidth
            local newH = startHeight
            local newX = startX
            local newY = startY
            for _, dir in directions do
                if dir == "Left" then
                    newW = startWidth - delta.X
                    newX = startX + delta.X
                elseif dir == "Right" then
                    newW = startWidth + delta.X
                elseif dir == "Top" then
                    newH = startHeight - delta.Y
                    newY = startY + delta.Y
                elseif dir == "Bottom" then
                    newH = startHeight + delta.Y
                end
            end
            if newW < minWidth then
                if table.find(directions, "Left") then newX = startX + (startWidth - minWidth) end
                newW = minWidth
            end
            if newH < minHeight then
                if table.find(directions, "Top") then newY = startY + (startHeight - minHeight) end
                newH = minHeight
            end
            ScreenGui_MainFrame.Size = UDim2.fromOffset(newW, newH)
            ScreenGui_MainFrame.Position = UDim2.fromOffset(newX, newY)
        end
    end)
    utility:CreateConnection(uis.InputEnded, function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            resizing = false
        end
    end)
end

        if Input.KeyCode == Window.Key then

            if Window.Enabled then
                -- Menü açılırken hemen input'u aktif et, sonra fade in
                ScreenGui.Enabled = true
                Window:Fade(true)
            else
                -- Menü kapanırken önce fade out yap, sonra input'u tamamen kapat
                Window:Fade(false)
                task.delay(0.3, function()

                        ScreenGui.Enabled = false
                    end
                end)
        end

            Window:Unload()
        end
    end)

local function UpdateSectionSizes(frame)
        if not frame then return end
        local sections = {}
        for _, child in ipairs(frame:GetChildren()) do
            if child:IsA("Frame") and child.BackgroundColor3 == Color3.fromRGB(40, 40, 40) then -- Section_Holder
                table.insert(sections, child)
            end
        end
        if #sections == 0 then return end
        local pad = frame:FindFirstChildWhichIsA("UIPadding")
        local layout = frame:FindFirstChildWhichIsA("UIListLayout")
        local topBottom = pad and (pad.PaddingTop.Offset + pad.PaddingBottom.Offset) or 0
        local inter = layout and (layout.Padding.Offset * (#sections - 1)) or 0
        local totalPadding = topBottom + inter
        local available = frame.AbsoluteSize.Y - totalPadding
        local minHeight = 120
        local eachHeight = math.max(available / #sections, minHeight)
        for _, sec in sections do
            sec.Size = UDim2.new(1, 0, 0, eachHeight)
        end
    end
    utility:CreateConnection(Page_Page_Left:GetPropertyChangedSignal("AbsoluteSize"), function()
        UpdateSectionSizes(Page_Page_Left)
    end)
    utility:CreateConnection(Page_Page_Right:GetPropertyChangedSignal("AbsoluteSize"), function()
        UpdateSectionSizes(Page_Page_Right)
    end)
    -- İlk açılışta çalıştır
    UpdateSectionSizes(Page_Page_Left)
    UpdateSectionSizes(Page_Page_Right)

end

if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
local relativeX = math.clamp((input.Position.X - frame.AbsolutePosition.X) / frame.AbsoluteSize.X, 0, 1)
local relativeY = math.clamp((input.Position.Y - frame.AbsolutePosition.Y) / frame.AbsoluteSize.Y, 0, 1)
indicator.Position = UDim2.new(relativeX, 0, relativeY, 0)
updateFunc(relativeX, relativeY)
end
end)

-- Store original properties per player
local originalProperties = {} -- Keyed by player
local function saveOriginalProperties(player)
	if player.Character then
		originalProperties[player] = originalProperties[player] or {}
		for partName, defaultProps in pairs(defaultProperties) do
			local part = player.Character:FindFirstChild(partName)
			if part then
				originalProperties[player][partName] = {
					Size = part.Size,
					CanCollide = part.CanCollide,
					Transparency = part.Transparency
				}
			else
				originalProperties[player][partName] = {
					Size = defaultProps.Size,
					CanCollide = defaultProps.CanCollide,
					Transparency = defaultProps.Transparency
				}
			end
		end
	end

local function startSilentAim()
	local players = game:GetService("Players")
	local plr = players.LocalPlayer
	local visualParts = {}
	local characterConnections = {}
	local heartbeatConnection = nil

	local function cleanup()
		if heartbeatConnection then
			heartbeatConnection:Disconnect()
			heartbeatConnection = nil
		end
		for player, connection in pairs(characterConnections) do
			if connection then
				connection:Disconnect()
				characterConnections[player] = nil
			end
		end
		for player, visualPart in pairs(visualParts) do
			if visualPart then
				visualPart:Destroy()
				visualParts[player] = nil
			end
			-- Orijinal boyutları geri yükle (sadece hitbox büyütülmüşse)
			if player.Character and originalProperties[player] then
				for partName, props in pairs(originalProperties[player]) do
					local part = player.Character:FindFirstChild(partName)
					if part then
						part.Size = props.Size
						part.CanCollide = props.CanCollide
						part.Transparency = props.Transparency
					end
				end
			end
		end
	end

	local function setupPlayer(player)
		if player.Name ~= plr.Name then
			saveOriginalProperties(player)
			characterConnections[player] = player.CharacterAdded:Connect(function()
				saveOriginalProperties(player)
				if visualParts[player] then
					visualParts[player]:Destroy()
					visualParts[player] = nil
				end
			end)
		end
	end

	for _, v in ipairs(players:GetPlayers()) do
		setupPlayer(v)
	end

	local playerAddedConnection = players.PlayerAdded:Connect(setupPlayer)
	local playerRemovingConnection = players.PlayerRemoving:Connect(function(player)
		if characterConnections[player] then
			characterConnections[player]:Disconnect()
			characterConnections[player] = nil
		end
		if visualParts[player] then
			visualParts[player]:Destroy()
			visualParts[player] = nil
		end
		originalProperties[player] = nil
	end)

	local lastToggleState = silentAimActive

	heartbeatConnection = game:GetService("RunService").Heartbeat:Connect(function()
	-- Toggle değiştiyse tüm visual'ları sil
		if lastToggleState ~= silentAimActive then
			for player, visualPart in pairs(visualParts) do
				if visualPart then
					visualPart:Destroy()
					visualParts[player] = nil
				end
			end
			lastToggleState = silentAimActive
		end

		-- Hitbox KAPALIYSA → orijinal boyutlara dön
		if not silentAimActive then
			for player, props in pairs(originalProperties) do
				if player.Character then
					for partName, partProps in pairs(props) do
						local part = player.Character:FindFirstChild(partName)
						if part then
							part.Size = partProps.Size
							part.CanCollide = partProps.CanCollide
							part.Transparency = partProps.Transparency
						end
					end
				end
			end
			return
		end

		-- Sadece düşmanlara uygula
		for _, v in ipairs(players:GetPlayers()) do
			if v.Name == plr.Name then continue end
			if not v.Character or not v.Character:FindFirstChild("HumanoidRootPart") then continue end

			local rootPart = v.Character.HumanoidRootPart

			-- ONLY ENEMIES: Takım arkadaşıysa HİÇBİR ŞEY YAPMA
			if ESPSettings.OnlyEnemies and v.Team == plr.Team then
			-- Hitbox büyütme, visual, hiçbir şey yapma
				continue
			end

			local fixedSize = hitboxxSize

			-- Hitbox genişlet
			for partName, _ in pairs(defaultProperties) do
				local part = v.Character:FindFirstChild(partName)
				if part then
					part.Size = Vector3.new(fixedSize, fixedSize, fixedSize)
					part.CanCollide = false
					part.Transparency = 1
				end
			end

			-- Visual Outline (HitboxVisual) - SADECE DÜŞMANLAR İÇİN
			local visualPart = visualParts[v]
			if visualPart and (not visualPart.Parent or visualPart.Parent ~= v.Character) then
				visualPart:Destroy()
				visualParts[v] = nil
			end

			if not visualParts[v] then
			-- Eski visual'ı temizle
				for _, existingPart in ipairs(v.Character:GetChildren()) do
					if existingPart.Name == "HitboxVisual" then
						existingPart:Destroy()
					end
				end

				visualPart = Instance.new("Part")
				visualPart.Name = "HitboxVisual"
				visualPart.Anchored = false
				visualPart.CanCollide = false
				visualPart.Transparency = 1
				visualPart.Size = Vector3.new(fixedSize, fixedSize, fixedSize)
				visualPart.Parent = v.Character

				local selectionBox = Instance.new("SelectionBox")
				selectionBox.Name = "VisualOutline"
				selectionBox.Adornee = visualPart
				selectionBox.Color3 = Color3.new(1, 0, 0)
				selectionBox.LineThickness = 0.05
				selectionBox.Transparency = 0
				selectionBox.Parent = visualPart

				local weld = Instance.new("Weld")
				weld.Name = "VisualWeld"
				weld.Part0 = rootPart
				weld.Part1 = visualPart
				weld.C0 = CFrame.new(0, 0, 0)
				weld.Parent = visualPart

				visualParts[v] = visualPart
			else
				visualPart.Size = Vector3.new(fixedSize, fixedSize, fixedSize)
			end
		end
	end)

	return function()
		cleanup()
		playerAddedConnection:Disconnect()
		playerRemovingConnection:Disconnect()
	end

-- Araç özelliklerini güncelleyen yardımcı fonksiyon
local function updateToolProperties(tool)
    if not tool:IsA("Tool") then return end
    -- RapidFire
    if rapidFireEnabled then
        local cooldown = tool:FindFirstChild("ShootingCooldown")
        if not cooldown then
            cooldown = Instance.new("NumberValue")
            cooldown.Name = "ShootingCooldown"
            cooldown.Parent = tool
        end
        if cooldown.Value ~= 0 then
            cooldown.Value = 0
        end
    end
    -- InfAmmo
    if infAmmoEnabled then
        local maxAmmo = tool:FindFirstChild("MaxAmmo")
        local ammo = tool:FindFirstChild("Ammo")
        if not maxAmmo then
            maxAmmo = Instance.new("NumberValue")
            maxAmmo.Name = "MaxAmmo"
            maxAmmo.Parent = tool
        end
        if not ammo then
            ammo = Instance.new("NumberValue")
            ammo.Name = "Ammo"
            ammo.Parent = tool
        end
        if maxAmmo.Value ~= 1000 then
            maxAmmo.Value = 1000
        end
        if ammo.Value ~= 1000 then
            ammo.Value = 1000
        end
    end
    -- GunRange
    if gunRangeEnabled then
        local range = tool:FindFirstChild("Range")
        if not range then
            range = Instance.new("NumberValue")
            range.Name = "Range"
            range.Parent = tool
        end
        if range.Value ~= 999999 then
            range.Value = 999999
        end
    end
end

-- İstatistikleri güncelleyen fonksiyon
local function updateStats()
    local char = player.Character
    if not char then return end
    -- Yeni araçları kontrol et ve güncelle
    local toolsToUpdate = {}
    for _, tool in ipairs(char:GetChildren()) do
        if tool:IsA("Tool") and not toolCache[tool] then
            toolCache[tool] = true
            table.insert(toolsToUpdate, tool)
        end
    end
    for _, tool in ipairs(player.Backpack:GetChildren()) do
        if tool:IsA("Tool") and not toolCache[tool] then
            toolCache[tool] = true
            table.insert(toolsToUpdate, tool)
        end
    end
    -- Önbelleği temizle
    for tool in pairs(toolCache) do
        if not tool.Parent then
            toolCache[tool] = nil
        end
    end
    -- Araçları güncelle
    for _, tool in ipairs(toolsToUpdate) do
        updateToolProperties(tool)
    end
end

-- Güncelleme döngüsünü başlatan fonksiyon
local function startUpdateLoop()
    if updateConnection then
        updateConnection:Disconnect()
        updateConnection = nil
    end
    if rapidFireEnabled or gunRangeEnabled or infAmmoEnabled then
        updateConnection = RunService.Heartbeat:Connect(function()
            if tick() - lastUpdate >= updateInterval then
                updateStats()
                lastUpdate = tick()
            end
        end)
        -- Araç eklenme/çıkarılma olaylarını dinle
        local toolAddedConnection
        local toolRemovedConnection
        toolAddedConnection = player.Backpack.ChildAdded:Connect(function(child)
            if child:IsA("Tool") then
                updateToolProperties(child)
                toolCache[child] = true
            end
        end)
        toolRemovedConnection = player.Backpack.ChildRemoved:Connect(function(child)
            if child:IsA("Tool") then
                toolCache[child] = nil
            end
        end)
        -- Karakter yeniden yüklendiğinde bağlantıları temizle ve döngüyü yeniden başlat
        local characterConnection
        characterConnection = player.CharacterAdded:Connect(function()
            toolCache = {}
            if toolAddedConnection then toolAddedConnection:Disconnect() end
            if toolRemovedConnection then toolRemovedConnection:Disconnect() end
            if updateConnection then updateConnection:Disconnect() end
            startUpdateLoop()
            characterConnection:Disconnect() -- Tek kullanımlık bağlantı
        end)
    end
end

-- Güncelleme döngüsünü durduran fonksiyon
local function stopUpdateLoop()
    if updateConnection then
        updateConnection:Disconnect()
        updateConnection = nil
    end
end

-- GlowMap: orijinalBasePart -> glowPart
local GlowMap = {}

-- Mevcut glow'ları temizle
local function ClearGlow(char)
    local glowFolder = char:FindFirstChild("FenasalGlowFolder")
    if glowFolder then
        for _, g in ipairs(glowFolder:GetChildren()) do
            for orig, glow in pairs(GlowMap) do
                if glow == g then
                    GlowMap[orig] = nil
                    break
                end
            end
            g:Destroy()
        end
        glowFolder:Destroy()
    end
end

-- Glow oluştur: BasePart’lar, aynı boyut ve pozisyonda
local function ApplyGlow(char)
    ClearGlow(char)

    local glowFolder = Instance.new("Folder")
    glowFolder.Name = "FenasalGlowFolder"
    glowFolder.Parent = char

    for _, part in pairs(char:GetChildren()) do
        if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then
            local glowPart = Instance.new("Part")
            glowPart.Name = "GlowPart"
            glowPart.Size = part.Size                       -- orijinal boyut
            glowPart.CFrame = part.CFrame                   -- orijinal pozisyon
            glowPart.Anchored = true
            glowPart.CanCollide = false
            glowPart.CanTouch = false
            glowPart.CanQuery = false
            glowPart.Material = Enum.Material.Neon
            glowPart.Color = SSSetting.GlowColor
            glowPart.Transparency = SSSetting.GlowIntensity
            glowPart.CastShadow = false
            glowPart.Parent = glowFolder

            GlowMap[part] = glowPart
        end
    end
end

                glowPart.Size = origPart.Size
            end)
        else
            if glowPart and glowPart.Parent then
                glowPart:Destroy()

            GlowMap[origPart] = nil
        end
    end

                    local be = targetChar:FindFirstChild("BodyEffects")
                    if not be then continue end

                    local ko = be:FindFirstChild("K.O")
                    local dead = be:FindFirstChild("Dead")
                    if not ko or not dead then continue end

                    -- 🛑 KRİTİK FİLTRE: Dead true ise K.O'ya bakmaksızın ES GEÇ
                    if dead.Value == true or ko.Value ~= true then
                        continue
                    end

                    local targetBody = targetChar:FindFirstChild("UpperTorso") or targetChar:FindFirstChild("Torso") or targetChar:FindFirstChild("HumanoidRootPart")

                    -- Void veya Body parça kontrolü
                    if not targetBody or targetBody.Position.Y <= VOID_LEVEL then
                        continue
                    end

                    isStomping = true

                    -- ===== STOMP DÖNGÜSÜ =====
                    while AutoStompEnabled do
                        -- 🛑 ANLIK KONTROLLER: Oyundan çıkma, Ölme, Kalkma veya Void
                        if not targetChar.Parent or dead.Value == true or ko.Value == false or targetBody.Position.Y <= VOID_LEVEL then
                            break
                        end

                        -- ✨ YÖNDEN BAĞIMSIZ TAKİP: 
                        -- Adamın CFrame'i yerine sadece Position'ını kullanıyoruz.
                        -- Böylece adam ters dönse bile sen hep yukarıda (Y + 2.5) kalırsın.
                        root.CFrame = CFrame.new(targetBody.Position + Vector3.new(0, 2.5, 0))
                        
                        -- Titremeyi ve savrulmayı önle
                        root.AssemblyLinearVelocity = Vector3.zero
                        root.AssemblyAngularVelocity = Vector3.zero

                        StompRemote:FireServer("Stomp")
                        task.wait()
                    end

                    -- Hedefle iş bittiğinde geri dön
                    isStomping = false
                    root.CFrame = lastSafePos
                    root.AssemblyLinearVelocity = Vector3.zero
                    root.AssemblyAngularVelocity = Vector3.zero
                    
                    task.wait(0.05)
                end

                task.wait(0.2)
            end
        end)
    end
})

-- GHOST (HAYALET) OLUŞTURMA FONKSİYONU
--------------------------------------------------------------------------------
local function CreateKillGhost(character)
    if not character or not character:FindFirstChild("HumanoidRootPart") then return end
    
    -- UI'dan gelen güncel rengi al
    local ghostColor = AnimSetting.anim_color or AnimSetting.KillAnimColor
    
    -- Hayalet modeli
    local ghostModel = Instance.new("Model")
    ghostModel.Name = "KillGhost_" .. character.Name
    ghostModel.Parent = workspace
    
    -- Karakter parçalarını kopyala
    for _, part in pairs(character:GetChildren()) do
        if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then
            part.Archivable = true
            local clone = part:Clone()

            clone:ClearAllChildren()
            clone.Anchored = true
            clone.CanCollide = false
            clone.CanQuery = false
            clone.CastShadow = false

            clone.Material = Enum.Material.Neon
            clone.Color = ghostColor
            clone.Transparency = 0
            clone.CFrame = part.CFrame
            clone.Parent = ghostModel
        end
    end

    -- Glow efekti
    local highlight = Instance.new("Highlight")
    highlight.Parent = ghostModel
    highlight.FillColor = ghostColor
    highlight.OutlineColor = Color3.new(1, 1, 1)
    highlight.FillTransparency = 0.4
    highlight.OutlineTransparency = 0

    --------------------------------------------------------------------------
    -- 3 SN BEKLE → SONRA SOLARAK KAYBOL
    --------------------------------------------------------------------------
    task.spawn(function()
        local waitBeforeFade = 3     -- Hayalet sabit kalma süresi
        local fadeDuration = 3       -- Solma süresi

        task.wait(waitBeforeFade)

        local tweenInfo = TweenInfo.new(
            fadeDuration,
            Enum.EasingStyle.Linear,
            Enum.EasingDirection.Out
        )

        for _, p in pairs(ghostModel:GetChildren()) do
            if p:IsA("BasePart") then
                TweenService:Create(p, tweenInfo, {
                    Transparency = 1
                }):Play()
            end
        end

        TweenService:Create(highlight, tweenInfo, {
            FillTransparency = 1,
            OutlineTransparency = 1
        }):Play()

        task.wait(fadeDuration)
        ghostModel:Destroy()
    end)
end

-- PLAYER MONITORING (K.O Takibi)
--------------------------------------------------------------------------------
local function MonitorPlayers()
    -- workspace.Players klasörünü bekle
    local playersFolder = workspace:WaitForChild("Players")

    local function setupKOListener(char)
        local bodyEffects = char:WaitForChild("BodyEffects", 15)
        if bodyEffects then
            local koValue = bodyEffects:WaitForChild("K.O", 15)
            if koValue then
                koValue:GetPropertyChangedSignal("Value"):Connect(function()
                    -- Sadece aktif olduğunda ve K.O true olduğunda çalışır
                    if koValue.Value == true and AnimSetting.Enable_killanim then
                        CreateKillGhost(char)
                    end
                end)
            end
        end
    end

    -- Yeni eklenen oyuncuları izle
    playersFolder.ChildAdded:Connect(function(child)
        if child:IsA("Model") then
            setupKOListener(child)
        end
    end)

    -- Mevcut oyuncuları tara
    for _, char in pairs(playersFolder:GetChildren()) do
        if char:IsA("Model") then
            setupKOListener(char)
        end
    end
end

local function createTrail()
    if not char then return end
    
    -- Parçaları tekrar bul (R6 ve R15 uyumlu)
    leftLeg = char:FindFirstChild("LeftFoot") or char:FindFirstChild("Left Leg")
    rightLeg = char:FindFirstChild("RightFoot") or char:FindFirstChild("Right Leg")

    if not leftLeg or not rightLeg then return end

    -- Eski trail varsa temizle
    if trail then 
        pcall(function() trail:Destroy() end) 
    end

    local att0 = Instance.new("Attachment")
    att0.Name = "TrailAttachment0"
    att0.Parent = leftLeg
    
    local att1 = Instance.new("Attachment")
    att1.Name = "TrailAttachment1"
    att1.Parent = rightLeg

    trail = Instance.new("Trail")
    trail.Name = "LegTrail"
    trail.Attachment0 = att0
    trail.Attachment1 = att1

    trail.Texture = "rbxassetid://1316045217"
    trail.TextureMode = Enum.TextureMode.Stretch
    trail.FaceCamera = true 
    trail.LightInfluence = 0 

    trail.Parent = char
end

    if not char or not char.Parent then

    if not trail or not trail.Parent or trail.Parent ~= char then

    if trail and trail.Parent then
        trail.Color = ColorSequence.new(PlayerSettings.TrailColor, PlayerSettings.TrailColor2)
        trail.Lifetime = PlayerSettings.TrailLifetime
        trail.WidthScale = NumberSequence.new(PlayerSettings.TrailWidth)
        trail.Transparency = NumberSequence.new(PlayerSettings.TrailTransparency, 1)
        trail.LightEmission = PlayerSettings.TrailGlow
    end
else
    if trail then
        trail:Destroy()
        trail = nil
    end
end

    -- 2) SELF CHAMS MANTIĞI (DÜZELTİLDİ)
    for _, part in pairs(char:GetChildren()) do
        if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then

        end
    end

local CenterDot = Instance.new("Frame", CursorMain)

CenterDot.ZIndex = 11
CenterDot.Visible = false

local rotation = 0

-- Render Loop (Mouse Takibi ve Animasyon)
RunService.RenderStepped:Connect(function(dt)
    if CrosshairSettings.Enabled then
        CursorMain.Visible = true
        UserInputService.MouseIconEnabled = false -- Gerçek imleci gizle
        
        -- Mouse Pozisyonunu Takip Et
        local mousePos = UserInputService:GetMouseLocation()
        CursorMain.Position = UDim2.new(0, mousePos.X, 0, mousePos.Y)

        -- Spin Hesaplama
        if CrosshairSettings.SpinEnabled then
            rotation = rotation + (dt * CrosshairSettings.SpinSpeed)
            CursorMain.Rotation = rotation
        else
            CursorMain.Rotation = 0
        end

        -- Pulse Hesaplama
        local currentGap = CrosshairSettings.Gap
        if CrosshairSettings.PulseEnabled then
            pulseVal = pulseVal + (dt * CrosshairSettings.PulseSpeed)
            currentGap = CrosshairSettings.Gap + (math.sin(pulseVal) * CrosshairSettings.PulseAmount)
        end

        -- Görsel Ayarlar
        local size = CrosshairSettings.Size
        local thick = CrosshairSettings.Thickness
        local color = CrosshairSettings.Color
        local mode = CrosshairSettings.Mode

        -- Mod Kontrolleri
        Lines[1].Visible = true -- Üst
        Lines[2].Visible = (mode ~= "T-Style") -- Alt
        Lines[3].Visible = true -- Sol
        Lines[4].Visible = true -- Sağ
        CenterDot.Visible = (mode == "Dot" or mode == "Circle")

        if mode == "Dot" then
            for i=1,4 do Lines[i].Visible = false end
            CenterDot.Size = UDim2.new(0, thick * 2, 0, thick * 2)
            CenterDot.BackgroundColor3 = color
            if not CenterDot:FindFirstChild("UICorner") then Instance.new("UICorner", CenterDot) end
        elseif mode == "Circle" then
            CenterDot.Size = UDim2.new(0, currentGap*2.5, 0, currentGap*2.5)
            CenterDot.BackgroundTransparency = 0.7
            CenterDot.BackgroundColor3 = color
            if not CenterDot:FindFirstChild("UICorner") then Instance.new("UICorner", CenterDot).CornerRadius = UDim.new(1,0) end
        end

        -- Çizgi Pozisyonları
        Lines[1].Size = UDim2.new(0, thick, 0, size)
        Lines[1].Position = UDim2.new(0, 0, 0, -currentGap - (size/2))
        
        Lines[2].Size = UDim2.new(0, thick, 0, size)
        Lines[2].Position = UDim2.new(0, 0, 0, currentGap + (size/2))
        
        Lines[3].Size = UDim2.new(0, size, 0, thick)
        Lines[3].Position = UDim2.new(0, -currentGap - (size/2), 0, 0)
        
        Lines[4].Size = UDim2.new(0, size, 0, thick)
        Lines[4].Position = UDim2.new(0, currentGap + (size/2), 0, 0)

        for i=1,4 do Lines[i].BackgroundColor3 = color end
    else
        CursorMain.Visible = false
        UserInputService.MouseIconEnabled = true -- Gerçek imleci geri getir
    end
end)

local shopItem = workspace.Ignored.Shop:GetChildren()[73]

local teleporting = false

local originalPos
local originalSpeed, originalJump, originalPlatformStand

UserInputService.InputBegan:Connect(function(input, gameProcessed)
	if gameProcessed then return end
	if input.KeyCode == Enum.KeyCode.O and not teleporting then
		teleporting = true

		local char = player.Character
		if char and char:FindFirstChild("HumanoidRootPart") and char:FindFirstChild("Humanoid") then
			local humanoid = char.Humanoid
			local hrp = char.HumanoidRootPart

			-- Eski değerleri kaydet
			originalPos = hrp.CFrame
			originalSpeed = humanoid.WalkSpeed
			originalJump = humanoid.JumpPower
			originalPlatformStand = humanoid.PlatformStand

			-- Karakteri sabitle
			humanoid.WalkSpeed = 0
			humanoid.JumpPower = 0
			humanoid.PlatformStand = true

			local targetCFrame = CFrame.new(shopItem.Head.Position - Vector3.new(0, 2, 0))
			local cooldown = 0.4
			local startTime = tick()

			-- FPS dostu sürekli teleport döngüsü
			while tick() - startTime < cooldown do
			-- Tek seferlik CFrame ataması, sürekli tween yok
				hrp.CFrame = targetCFrame

				-- FireClick’leri küçük gruplar halinde yolla
				for i = 1, 15 do
					fireclickdetector(shopItem.ClickDetector)
				end

				task.wait(0.2)
			end

		
			hrp.CFrame = originalPos

			
			humanoid.WalkSpeed = originalSpeed
			humanoid.JumpPower = originalJump
			humanoid.PlatformStand = originalPlatformStand
		end

		teleporting = false
	end
end)
local maxSpeed = 325 -- velocidade máxima
local enabled = false

local player = game.Players.LocalPlayer
local UserInputService = game:GetService("UserInputService")

-- Detectar pressionamento e liberação da tecla X
UserInputService.InputBegan:Connect(function(input, gameProcessed)
	if gameProcessed then return end
	if input.KeyCode == Enum.KeyCode.X then
		enabled = true
	end
end)

UserInputService.InputEnded:Connect(function(input, gameProcessed)
	if gameProcessed then return end
	if input.KeyCode == Enum.KeyCode.X then
		enabled = false
		-- Resetar velocidade ao padrão
		if player.Character and player.Character:FindFirstChildOfClass("Humanoid") then
			player.Character:FindFirstChildOfClass("Humanoid").WalkSpeed = 16
		end
	end
end)

-- Loop para aplicar speed máxima
game:GetService("RunService").RenderStepped:Connect(function()
	if enabled and player.Character and player.Character:FindFirstChildOfClass("Humanoid") then
		player.Character:FindFirstChildOfClass("Humanoid").WalkSpeed = maxSpeed
	end
end)
